cmake_minimum_required(VERSION 3.16)
project(seminar10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /WX)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(seminar10_lib src/task1.cpp src/task2.cpp src/task3.cpp src/app.cpp)
target_include_directories(seminar10_lib PUBLIC include)
add_executable(seminar10 src/main.cpp)
target_link_libraries(seminar10 PRIVATE seminar10_lib)
add_executable(manual_tests tests/manual_tests.cpp)
target_link_libraries(manual_tests PRIVATE seminar10_lib)
enable_testing()
add_test(NAME seminar10_manual_tests COMMAND manual_tests)
